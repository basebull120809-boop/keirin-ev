(function (global, $) { // ===== 定数定義 ===== function createCampaignCommon(deps) { deps = deps || {}; var ajax = deps.ajax || $.ajax; var apiURL = deps.apiURL || ""; var timeout = deps.timeout || 15000; var showBadgeId = deps.showBadgeId; // ========================= // UI（共通振る舞い） // ========================= function showCampaignBadge(data) { for(var i = 0; i < data.length; i++) { var badgeId = showBadgeId + data[i]; $('#' + badgeId).show(); } } // ========================= // API共通（Promise） // ========================= function normalizeError(jqXHR, textStatus) { var message = 'エラーが発生しました(code: ' + jqXHR.status + ')'; if (textStatus === 'timeout') message = 'エラーが発生しました(code: 504)'; return { status: (jqXHR && jqXHR.status) || 0, message: message, raw: jqXHR }; } function request(opts) { var dfd = $.Deferred(); ajax({ type: opts.method || 'POST', url: apiURL, data: opts.data, dataType: opts.dataType || 'json', timeout: timeout }) .done(function (data) { dfd.resolve(data); }) .fail(function (jqXHR, textStatus) { dfd.reject(normalizeError(jqXHR, textStatus)); }); return dfd.promise(); } // ========================= // API（機能） // ========================= function getCampaign(payload) { var method = payload.method || 'GET'; return request({ method: method, data: {} }); } // ========================= // 外に返すもの // ========================= return { api: { getCampaign: getCampaign, }, ui: { showCampaignBadge: showCampaignBadge, } }; } global.createCampaignCommon = createCampaignCommon; })(window, jQuery);
